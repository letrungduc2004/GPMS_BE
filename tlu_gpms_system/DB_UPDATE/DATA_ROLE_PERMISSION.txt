-- 5.2 Teacher: giáo viên trong trường thuộc đúng ngành mình làm trưởng, số lượng hướng dẫn mặc định 20
INSERT INTO teacher (public_id, organization, amount, major_id, account_id)
SELECT UUID_TO_BIN(UUID()), 'Trường Đại học Thủy lợi', 20, m.id, a.id
FROM major m
JOIN account a ON a.first_name = 'Nguyễn' AND a.last_name = m.name
WHERE m.code IN ('KTPM01', 'HTTT01', 'CNTT01', 'ANM01', 'AI01');

-- 5.3 Role: SUPERVISOR (như giáo viên import) + HEAD_OF_MAJOR
INSERT INTO user_role (user_id, role_name)
SELECT a.id, r.role_name
FROM major m
JOIN account a ON a.first_name = 'Nguyễn' AND a.last_name = m.name
CROSS JOIN (SELECT 'SUPERVISOR' AS role_name UNION ALL SELECT 'HEAD_OF_MAJOR') r
WHERE m.code IN ('KTPM01', 'HTTT01', 'CNTT01', 'ANM01', 'AI01');

-- 5.4 Gán trưởng ngành cho từng ngành
-- MySQL Workbench bật safe update mode (lỗi 1175 vì WHERE không lọc theo cột key),
-- nên tắt tạm trong phiên này rồi bật lại
SET SQL_SAFE_UPDATES = 0;
UPDATE major m
JOIN account a ON a.first_name = 'Nguyễn' AND a.last_name = m.name
SET m.head_account_id = a.id
WHERE m.code IN ('KTPM01', 'HTTT01', 'CNTT01', 'ANM01', 'AI01');
SET SQL_SAFE_UPDATES = 1;

COMMIT;
